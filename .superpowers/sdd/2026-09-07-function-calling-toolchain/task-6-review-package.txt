+def parse_sse_events(sse_text: str):
+    """解析 SSE 响应文本为事件列表和 done 标记"""
+    events = []
+    has_done = False
+    for line in sse_text.splitlines():
+        line = line.strip()
+        if not line:
+            continue
+        if line.startswith("data:"):
+            data_str = line[len("data:"):].strip()
+            if data_str == "[DONE]":
+                has_done = True
+            else:
+                events.append(json.loads(data_str))
+    return events, has_done
+
